Program
       ::= Body

Body
       ::= VarDecl* Block

VarDecl
       ::= Type Identifier ( ',' Identifier )* ';'

Block
       ::= '{' Stmt+ '}'

Stmt
       ::= 'if' '(' Expr ')' Block 'else' Block
         | 'while' '(' Expr ')' Block
         | 'break' ';'
         | Var '=' Expr ';'
         | ';'

Expr
       ::= '(' Expr '?' Expr ':' Expr ')'
         | '(' Expr Binop Expr ')'
         | '(' Unop Expr ')'
         | '(' Expr ')'
         | Var
         | Literal

Binop
       ::= '+'
         | '-'
         | '*'
         | '/'
         | '%'
         | '&'
         | '|'
         | '<'
         | '>'
         | '='

Unop
       ::= '~'
         | '!'

Type
       ::= 'int'
         | 'bool'
         | 'String'

Literal
       ::= Num
         | 'true'
         | 'false'
         | String

Var
       ::= Identifier

Num
       ::= [0-9]+

String
       ::= '"' PrintableASCII* '"'

PrintableASCII
       ::= [#x20-#x7E]

Identifier
       ::= [a-zA-Z] [a-zA-Z0-9_]*
